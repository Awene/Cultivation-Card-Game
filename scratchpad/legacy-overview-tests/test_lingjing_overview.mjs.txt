// ARCHIVE ONLY: pre-unification renderer and obsolete mock API; not executable verification.
// 本地模拟酒馆变量与聊天 API，执行总览中的真实 EJS JavaScript 块及原文输出表达式。
// 不声称替代 SillyTavern / ST-Prompt-Template 的浏览器端联调。
import { readFileSync } from 'node:fs';
import assert from 'node:assert/strict';
import vm from 'node:vm';
import { build, data, targetPath } from './build_lingjing_overview.mjs';

const source = readFileSync(targetPath, 'utf8').replace(/\r\n/g, '\n').trim();
assert.equal(source, build().trim(), '总览必须与蓝图及构建逻辑一致');
let program = 'let rendered = "";\n';
const tag = /<%([_=\-]?)([\s\S]*?)([_\-]?)%>/g;
let cursor = 0;
for (const m of source.matchAll(tag)) {
  assert(!source.slice(cursor, m.index).trim(), '仅允许本总览使用的块间空白');
  program += (m[1] === '-' || m[1] === '=') ? `rendered += (${m[2]});\n` : m[2] + '\n';
  cursor = m.index + m[0].length;
}
assert(!source.slice(cursor).trim());
program += '\nrendered;';
const compiled = new vm.Script(program, { filename: targetPath });
function render(vars = {}, messages = [], historyMode = 'normal') {
  const values = { '地点.世界': '灵界', '地点.地域': '灵境大陆', '地点.具体地点': '', ...vars };
  const ctx = { getMessageVar: key => values[key.replace(/^stat_data\./, '')] };
  if (historyMode !== 'missing') ctx.getChatMessages = () => {
    if (historyMode === 'throw') throw new Error('模拟无聊天权限');
    return messages;
  };
  return compiled.runInNewContext(ctx, { timeout: 3000 });
}
let tests = 0;
function test(name, fn) { fn(); tests++; console.log('通过: ' + name); }

});

});
test('他大陆只保留生态、三大宗和两处重要秘境', () => {
  const out = render({'地点.地域':'星坠大陆','地点.具体地点':'留春旧苑'}, ['织雨藏楼']);
  for (const e of data.ecology) assert(out.includes(e.title));

  assert(!out.includes('### 素问山'));
  assert(!out.includes('## 灵境大陆凡国与城市'));
  assert(!out.includes('<region_map'));
  assert(!out.includes('- 1. 停候花径'));
});

});

test('32个人物都可显式展开；出场后不覆盖初始境界、外貌和关系', () => {
  for (const c of Object.values(data.characters)) {
    const fresh = render({}, [c.name]);
    assert(fresh.includes('### ' + c.name));
    assert(fresh.includes(c.fields['性格与职责']));
    const seen = render({'关系列表':{[c.name]:{境界:'变化后'}}}, [c.name]);
    assert(seen.includes('### ' + c.name));
    assert(!seen.includes(c.fields['形貌与衣饰']), c.name+'形貌重复');
    assert(!seen.includes(c.fields['性格与职责']), c.name+'初始档案重复');
    assert(!seen.includes(c.realm + '；' + c.role), c.name+'境界重复');
    for (const line of data.relations) {
      if (line.split('：')[0].includes(c.name)) assert(!seen.includes(line), c.name+'旧关系覆盖');
    }
  }
});
