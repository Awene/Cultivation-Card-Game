const source = readFileSync(targetPath, "utf8").replace(/\r\n/g, "\n");
assert.equal(source, build(), "总览必须与蓝图构建完全一致");
let program = 'let rendered="";\n',
  cursor = 0;
for (const m of source.matchAll(/<%([_=\-]?)([\s\S]*?)([_\-]?)%>/g)) {
  assert(!source.slice(cursor, m.index).trim());
  program +=
    (["-", "="].includes(m[1]) ? "rendered+=(" + m[2] + ");" : m[2]) + "\n";
  cursor = m.index + m[0].length;
}
assert(!source.slice(cursor).trim());
program += "\nrendered;";
const compiled = new vm.Script(program, { filename: targetPath });
const calls = [];
function render(vars = {}, messages = [], mode = "normal") {
  const values = {
    "地点.世界": "灵界",
    "地点.地域": "沧溟大陆",
    "地点.具体地点": "",
    ...vars,
  };
  const ctx = { getMessageVar: (k) => values[k.replace(/^stat_data\./, "")] };
  if (mode !== "missing")
    ctx.getChatMessages = (range) => {
      calls.push(range);
      if (mode === "throw") throw Error("模拟消息API不可用");
      const normalized = messages.map((m, i) =>
        typeof m === "string"
          ? { message: m, message_id: i }
          : { ...m, message_id: i },
      );
      if (range === -1) return normalized.slice(-1);
      const [start, end] = String(range).split("-").map(Number);
      return normalized.filter(
        (m) => m.message_id >= start && m.message_id <= end,
      );
    };
  return compiled.runInNewContext(ctx, { timeout: 5000 });
}

test("八国可按名字、城市、人物和地标展开", () => {
  const base = render();
  for (const c of data.countries) {
    assert(!base.includes(c.fields["行政与军事"]));
    const site = data.cities.find((s) => s.country === c.name);

  }
});

    ),
  );
});
