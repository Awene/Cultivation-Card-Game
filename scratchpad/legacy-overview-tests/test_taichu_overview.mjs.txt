function compile(tpl) {
  let code = 'let __out="";\n', cursor = 0;
  const re = /<%([_=\-]?)([\s\S]*?)([_\-]?)%>/g;
  for (const m of tpl.matchAll(re)) {
    code += `__out += ${JSON.stringify(tpl.slice(cursor,m.index))};\n`;
    code += ['=','-'].includes(m[1]) ? `__out += String((${m[2]}) ?? '');\n` : m[2] + '\n';
    cursor = m.index + m[0].length;
  }
  return code + `__out += ${JSON.stringify(tpl.slice(cursor))};\n__out;`;
}

  return program.runInNewContext({ getMessageVar:p => vars[p], getChatMessages:() => {

}
